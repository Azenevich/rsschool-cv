## 🖥️ Code example (JavaScript):
```
function strCount(obj){
  let result = 0;
  for (let key in obj) {
    const value = obj[key];

    if (typeof value === "string") {
      result++;
    }
    else if (typeof value === 'object' && value !== null) {
      result += strCount(value)
    }
  }
  return result;
}
```
